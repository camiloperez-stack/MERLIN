    <!-- Script Principal -->

    <script>

        const stages = [

            { id: 'contacto', title: '1. Contacto', color: 'border-blue-500' },

            { id: 'tecnico', title: '2. Técnico', color: 'border-yellow-500' },

            { id: 'comercial', title: '3. Comercial', color: 'border-orange-500' },

            { id: 'financiero', title: '4. Financiero', color: 'border-emerald-500' },

            { id: 'radicacion', title: '5. Radicación', color: 'border-purple-500' }

        ];


        let clients = JSON.parse(localStorage.getItem('merlin_clients')) || [

            {

                id: 1,

                nombre: 'Carlos Pérez',

                cedula: '1018239120',

                telefono: '3001234567',

                ramo: 'Autos',

                origen: 'Meta',

                placa: 'KLR-456',

                marca: 'Mazda 3',

                aseguradora: 'Sura',

                primaBruta: 2500000,

                stage: 'contacto',

                substate: 'Potencial'

            }

        ];


        function renderPipeline() {

            const container = document.getElementById('pipeline-container');

            container.innerHTML = '';


            stages.forEach(stage => {

                const column = document.createElement('div');

                column.className = `w-72 bg-slate-200/60 rounded-lg p-3 flex flex-col gap-3 border-t-4 ${stage.color} min-h-[70vh]`;

                

                const stageClients = clients.filter(c => c.stage === stage.id);


                column.innerHTML = `

                    <div class="flex justify-between items-center mb-1">

                        <h2 class="font-bold text-sm text-slate-700 uppercase">${stage.title}</h2>

                        <span class="bg-slate-300 text-slate-700 text-xs px-2 py-0.5 rounded-full font-bold">${stageClients.length}</span>

                    </div>

                    <div class="flex-1 space-y-3 overflow-y-auto" ondragover="allowDrop(event)" ondrop="drop(event, '${stage.id}')">

                        ${stageClients.map(c => createCard(c)).join('')}

                    </div>

                `;

                container.appendChild(column);

            });

        }


        function createCard(c) {

            return `

                <div id="card-${c.id}" draggable="true" ondragstart="drag(event, ${c.id})" class="bg-white p-3 rounded-lg shadow-sm border border-slate-200 hover:shadow-md transition cursor-grab active:cursor-grabbing">

                    <div class="flex justify-between items-start mb-2">

                        <span class="text-[10px] font-bold px-2 py-0.5 bg-indigo-50 text-indigo-700 rounded-full">${c.ramo}</span>

                        <span class="text-[10px] text-slate-400 font-medium">${c.origen}</span>

                    </div>

                    <h3 class="font-bold text-sm text-slate-800">${c.nombre}</h3>

                    <p class="text-xs text-slate-500">CC: ${c.cedula}</p>

                    ${c.placa ? `<p class="text-xs font-semibold text-slate-700 mt-1"><i class="fa-solid fa-car text-slate-400 mr-1"></i>${c.placa} (${c.marca})</p>` : ''}

                    <div class="mt-3 pt-2 border-t flex justify-between items-center text-xs">

                        <span class="text-emerald-600 font-bold">$${Number(c.primaBruta || 0).toLocaleString()}</span>

                        <select onchange="changeStage(${c.id}, this.value)" class="text-[11px] bg-slate-100 border rounded px-1 py-0.5">

                            ${stages.map(s => `<option value="${s.id}" ${c.stage === s.id ? 'selected' : ''}>${s.title}</option>`).join('')}

                        </select>

                    </div>

                </div>

            `;

        }


        function toggleModal() {

            document.getElementById('modal').classList.toggle('hidden');

        }


        document.getElementById('client-form').addEventListener('submit', function(e) {

            e.preventDefault();

            const newClient = {

                id: Date.now(),

                nombre: document.getElementById('nombre').value,

                cedula: document.getElementById('cedula').value,

                telefono: document.getElementById('telefono').value,

                ciudad: document.getElementById('ciudad').value,

                ramo: document.getElementById('ramo').value,

                origen: document.getElementById('origen').value,

                placa: document.getElementById('placa').value,

                marca: document.getElementById('marca').value,

                modelo: document.getElementById('modelo').value,

                aseguradora: document.getElementById('aseguradora').value,

                primaBruta: document.getElementById('primaBruta').value,

                primaNeta: document.getElementById('primaNeta').value,

                stage: 'contacto'

            };

            clients.push(newClient);

            saveAndRender();

            toggleModal();

            this.reset();

        });


        function changeStage(id, newStage) {

            clients = clients.map(c => c.id === id ? { ...c, stage: newStage } : c);

            saveAndRender();

        }


        // Drag & Drop

        function allowDrop(ev) { ev.preventDefault(); }

        function drag(ev, id) { ev.dataTransfer.setData("text", id); }

        function drop(ev, stageId) {

            ev.preventDefault();

            const id = ev.dataTransfer.getData("text");

            changeStage(Number(id), stageId);

        }


        function saveAndRender() {

            localStorage.setItem('merlin_clients', JSON.stringify(clients));

            renderPipeline();

        }


        renderPipeline();
